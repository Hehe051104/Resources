import numpy as np
#属性1和2的属性值数组
x_1 = [1,2,3,4,5]
x_2 = [6,7,8,9,10]
x_1 = np.asarray(x_1)
x_2 = np.asarray(x_2)

x_1_sum = 0
x_2_sum = 0

####begin####
#计算均值存于变量x_1_mean/x_2_mean
x_1_mean = np.mean(x_1)
x_2_mean = np.mean(x_2)
####end####

x_1_conv = 0
x_2_conv = 0
x_1_2_cov = 0

####begin####
#计算方差与协方差
#x1方差存于x_1_conv/x2方差存于x_2_conv/协方差存于x_1_2_cov
x_1_conv = np.var(x_1)  # 总体方差
x_2_conv = np.var(x_2)  # 总体方差
x_1_2_cov = np.cov(x_1, x_2)[0,1]  # 总体协方差
####end####

#输出协方差矩阵
x = np.asarray([[x_1_conv,x_1_2_cov],
     [x_1_2_cov,x_2_conv]])
print(x)
