import math


def quantile_p(data, p):
    # 对列表元素进行排序
    data.sort()

    # 计算指数i
    i = len(data) * p

    # 判断是否为整数
    if str(i)[-1] == '0':
        ########## Begin ##########
        # 整数处理方式，(x[i-1]+x[i])/2
        idx = int(i)
        res = (data[idx-1] + data[idx]) / 2
        ########## End ##########
    else:
        ########## Begin ##########
        # 非整数处理方式，x[ceil(i)]
        idx = math.ceil(i) - 1
        res = data[idx]
        ########## End ##########
    return res


n = float(input())
# 创建数据
data = [36, 40, 6, 7, 49, 15, 39, 41, 42, 43, 47]
q = quantile_p(data, n)
print("Q:", q)
