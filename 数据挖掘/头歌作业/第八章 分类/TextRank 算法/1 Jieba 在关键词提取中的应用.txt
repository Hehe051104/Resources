import jieba.analyse
sentence = input()
# 任务：基于jieba中的TF-IDF算法完成对sentence的关键词提取，提取前三个关键词并以一行输出

# ********** Begin *********#
keywords = jieba.analyse.extract_tags(sentence, topK=3, withWeight=True, allowPOS=('n', 'nr', 'ns'))
for item in keywords:
    print(item[0],end=" ")
# ********** End **********#
