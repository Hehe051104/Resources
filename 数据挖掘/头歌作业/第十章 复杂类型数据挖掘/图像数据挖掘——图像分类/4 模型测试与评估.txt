import pandas as pd
import numpy as np
import tensorflow as tf
import os
from keras.utils import to_categorical
from keras.models import Sequential
from keras.layers.core import Dense, Activation
from keras.optimizers import Adam
from sklearn.metrics import accuracy_score, recall_score
import keras

# 构建测试集
test_data = np.load('./test_data.npz', allow_pickle=True)
x_test = test_data["x_test"]
y_test = test_data["y_test"]

# 确保标签是数值类型
y_test = y_test.astype(int)

# 加载模型
new_model = keras.models.load_model('./my_model.h5')

# 预测
predictions = new_model.predict(x_test)
test_prediction = np.argmax(predictions, axis=1)

#/*************** Begin***************/
# 要求：
# 请补全api中的参数，使最终的accuracy>0.8,recall>0.8
# 步骤1：计算准确率（预测为猫中真正的猫的概率）
accuracy = accuracy_score(y_test, test_prediction)

# 步骤2：计算召回率（所有猫集合中预测为猫的概率）
recall = recall_score(y_test, test_prediction, average='binary')

#/*************** end***************/

# 测试
print(accuracy, recall)
