import myMatrix
n = int(input()) #杨辉三角总层数
########## Begin ##########
A = [[0] * n for _ in range(n)]

for i in range(n):
    A[i][0] = 1  # 首列全为1

for i in range(1, n):
    for j in range(1, i + 1):  # 第 i 行只计算到第 i 列
        A[i][j] = A[i-1][j-1] + A[i-1][j]
########## End ##########
myMatrix.printMatrix(A)          #打印矩阵A
print('-'*n*6)                   #打印分隔线
myMatrix.printMatrixAsYangHui(A) #打印杨辉三角
