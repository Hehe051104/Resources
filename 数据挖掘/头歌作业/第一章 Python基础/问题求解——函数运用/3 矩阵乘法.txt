def multiply(A, B):
    x,y=len(A),len(B[0])
    li=[[0 for _ in range(y)] for _ in range(x)]
    for i in range(len(A)):
        for j in range(len(B[0])):
            for k in range(len(B)):
                li[i][j]+=A[i][k]*B[k][j]
    return li
