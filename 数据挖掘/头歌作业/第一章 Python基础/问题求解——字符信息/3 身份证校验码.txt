li=[7,9,10,5,8,4,2,1,6,3,7,9,10,5,8,4,2]
le=[1,0,'X',9,8,7,6,5,4,3,2]
def checksum(id):
    s=0
    for i in range(len(id)):
        s+=int(id[i])*li[i]
    return le[s%11]
########## End ##########

ID = input()      #身份证前17位
cs = checksum(ID) #求校验码
print('%s -> %s' % (ID, cs))
