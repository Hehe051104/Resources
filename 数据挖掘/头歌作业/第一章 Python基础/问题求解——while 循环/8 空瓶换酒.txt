n = int(input())
########## Begin ##########
k = n
g = n

while k >= 2 or g >= 4:
    from_bottle = k // 2
    from_cap = g // 4
    total_new = from_bottle + from_cap
    
    if total_new == 0:
        break

    n += total_new

    k = k % 2 + total_new
    g = g % 4 + total_new

print(n)
########## End ##########
